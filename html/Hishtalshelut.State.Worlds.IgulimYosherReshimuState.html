module Hishtalshelut.State.Worlds.IgulimYosherReshimuState where

open import Data.List using (List; []; _∷_; concatMap; map)
open import Data.Product using (_×_; _,_)
open import Hishtalshelut.Domain.Worlds.IgulimYosherReshimu using (ReshimuSpec; seph; vesselInner; vesselOuter; toReshimuSpec)
open import Hishtalshelut.Domain.Worlds.IgulimYosher using (World; allWorlds; worldToId; Partzuf; allPartzufs; partzufToId; OlamId; PartzufId; allSefirotList; sefirahUnit)

-- | State of Reshimu imprints before Kav enters
record ReshimuState : Set where
  field
    reshimuByPartzuf : List (OlamId × PartzufId × List ReshimuSpec)

open ReshimuState public

-- | Initial ReshimuState: imprint list per Partzuf/Olam from all SefirahUnits
initialReshimuState : ReshimuState
initialReshimuState = record
  { reshimuByPartzuf =
      concatMap (λ w →
        let ol    = worldToId w
        in  map (λ p →
              let pid   = partzufToId w p
                  units = map sefirahUnit allSefirotList
                  specs = map toReshimuSpec units
              in (ol , pid , specs))
            allPartzufs)
      allWorlds
  }

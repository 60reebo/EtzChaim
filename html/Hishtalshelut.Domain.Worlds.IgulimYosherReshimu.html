--------------------------------------------------
-- ReshimuSpec (Domain Layer for IgulimYosher)
--------------------------------------------------
module Hishtalshelut.Domain.Worlds.IgulimYosherReshimu where

open import Data.Nat using (ℕ)
open import Hishtalshelut.Domain.Worlds.IgulimYosher using (Sefirah; VesselKind; SefirahUnit; CircleSpec)
open SefirahUnit public
open CircleSpec public

-- | Reshimu imprint specification per seph (before Kav enters)
record ReshimuSpec : Set where
  field
    seph        : Sefirah
    vesselInner : VesselKind
    vesselOuter : VesselKind
    purity      : ℕ

-- | Convert a SefirahUnit to a Reshimu imprint spec
toReshimuSpec : SefirahUnit → ReshimuSpec
toReshimuSpec su = record
  { seph        = seph (circle su)
  ; vesselInner = vesselInner (circle su)
  ; vesselOuter = vesselOuter (circle su)
  ; purity      = 0
  }
